'use strict';
var ORT_CDN = 'https://cdn.jsdelivr.net/npm/onnxruntime-web@1.17.3/dist/ort.min.js';
var session = null;
var IMG_SIZE = 256;
function post(type,extra){ self.postMessage(Object.assign({type:type},extra||{})); }

// IndexedDB cache
function openDB(){return new Promise(function(res,rej){var r=indexedDB.open('nezumi-model-cache',1);r.onupgradeneeded=function(e){e.target.result.createObjectStore('models');};r.onsuccess=function(e){res(e.target.result);};r.onerror=function(e){rej(e.target.error);};});}
function dbGet(db,key){return new Promise(function(res,rej){var tx=db.transaction('models','readonly'),r=tx.objectStore('models').get(key);r.onsuccess=function(){res(r.result);};r.onerror=function(){rej(r.error);};});}
function dbPut(db,key,val){return new Promise(function(res,rej){var tx=db.transaction('models','readwrite'),r=tx.objectStore('models').put(val,key);r.onsuccess=function(){res();};r.onerror=function(){rej(r.error);};});}

async function fetchModel(url){
  var db;try{db=await openDB();}catch(e){db=null;}
  if(db){try{var c=await dbGet(db,url);if(c){post('progress',{pct:95,label:'キャッシュから読み込み中…'});return c;}}catch(e){}}
  post('progress',{pct:0,label:'モデルをダウンロード中…'});
  var resp=await fetch(url);
  if(!resp.ok) throw new Error('fetch failed: '+resp.status);
  var reader=resp.body.getReader(), total=parseInt(resp.headers.get('content-length')||'0',10), chunks=[], received=0;
  while(true){var _=await reader.read();if(_.done)break;chunks.push(_.value);received+=_.value.length;if(total>0)post('progress',{pct:Math.round(received/total*90),label:Math.round(received/1048576)+' / '+Math.round(total/1048576)+' MB'});}
  var buf=new Uint8Array(received),off=0;chunks.forEach(function(c){buf.set(c,off);off+=c.length;});
  if(db){try{await dbPut(db,url,buf.buffer);}catch(e){}}
  return buf.buffer;
}

async function init(msg){
  try{
    importScripts(ORT_CDN);
    post('status',{state:'busy',text:'onnxruntime-web 読み込み完了'});
    var buf=await fetchModel(msg.modelUrl);
    IMG_SIZE = msg.imgSize || IMG_SIZE;
    post('progress',{pct:95,label:'セッション初期化中…'});
    var eps=['wasm'];
    if(msg.preferWebGPU){try{if(navigator.gpu||typeof GPUAdapter!=='undefined') eps=['webgpu','wasm'];}catch(e){}}
    var ortBase = 'https://cdn.jsdelivr.net/npm/onnxruntime-web@1.17.3/dist/';
    ort.env.wasm.wasmPaths   = ortBase;
    ort.env.wasm.numThreads  = msg.threads || 2;
    ort.env.wasm.simd        = true;
    session = await ort.InferenceSession.create(buf, { executionProviders: eps, graphOptimizationLevel: 'all' });
    post('ready',{ep:eps[0]});
  }catch(e){ post('error',{text:String(e.message||e)}); }
}

// fp16 bit-pattern (Uint16Array) -> Float32Array
function decodeF16(u16){
  var n=u16.length,f32=new Float32Array(n);
  for(var i=0;i<n;i++){
    var h=u16[i],s=(h>>>15)?-1:1,e=(h>>>10)&0x1f,m=h&0x3ff;
    if(e===0)f32[i]=s*Math.pow(2,-14)*(m/1024);
    else if(e===31)f32[i]=m?NaN:s*Infinity;
    else f32[i]=s*Math.pow(2,e-15)*(1+m/1024);
  }
  return f32;
}

async function run(msg){
  try{
    var S=msg.size||IMG_SIZE, isF16=(msg.dtype==='float16');
    var imgF32  = isF16 ? decodeF16(new Uint16Array(msg.imgArr))  : new Float32Array(msg.imgArr);
    var maskF32 = isF16 ? decodeF16(new Uint16Array(msg.maskArr)) : new Float32Array(msg.maskArr);
    var imgT =new ort.Tensor('float32',imgF32, [1,3,S,S]);
    var maskT=new ort.Tensor('float32',maskF32,[1,1,S,S]);
    var feeds={};feeds[session.inputNames[0]]=imgT;feeds[session.inputNames[1]]=maskT;
    var t0=performance.now(), res=await session.run(feeds), elapsed=performance.now()-t0;
    var out=res[session.outputNames[0]].data, n=S*S, rgba=new Uint8ClampedArray(n*4);
    var outMax=0;
    for(var t=0;t<out.length;t++){ if(out[t]>outMax) outMax=out[t]; }
    var scale = (outMax <= 1.01) ? 255 : 1;
    for(var i=0;i<n;i++){
      rgba[i*4+0]=Math.max(0,Math.min(255,out[i    ]*scale))|0;
      rgba[i*4+1]=Math.max(0,Math.min(255,out[n+i  ]*scale))|0;
      rgba[i*4+2]=Math.max(0,Math.min(255,out[2*n+i]*scale))|0;
      rgba[i*4+3]=255;
    }
    var minR=255,maxR=0,minG=255,maxG=0,minB=255,maxB=0;
    for(var k=0;k<n;k++){
      var r=rgba[k*4+0], g=rgba[k*4+1], b=rgba[k*4+2];
      if(r<minR)minR=r; if(r>maxR)maxR=r;
      if(g<minG)minG=g; if(g>maxG)maxG=g;
      if(b<minB)minB=b; if(b>maxB)maxB=b;
    }
    post('result',{rgba:rgba,ep:session.handler&&session.handler._ep||'wasm',elapsedMs:elapsed,stats:{minR:minR,maxR:maxR,minG:minG,maxG:maxG,minB:minB,maxB:maxB}});
  }catch(e){ post('error',{text:String(e.message||e)}); }
}

self.onmessage=function(e){var m=e.data;if(m.type==='init')init(m);if(m.type==='run')run(m);};
